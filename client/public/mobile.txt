    body { font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, Helvetica, Arial, sans-serif; background: #f5f5f5; padding: 16px; }
    .card { background: white; border-radius: 12px; padding: 16px; margin-bottom: 12px; box-shadow: 0 2px 8px rgba(0,0,0,0.05); }
    .search-box { width: 100%; padding: 12px; border: 1px solid #ddd; border-radius: 10px; font-size: 16px; margin-bottom: 16px; }
    .title { font-size: 20px; font-weight: bold; margin-bottom: 16px; }
    .amount { font-weight: bold; color: #e53935; }
    .text-muted { color: #888; font-size: 12px; }
    .flex-between { display: flex; justify-content: space-between; align-items: center; }
    .loading { text-align: center; padding: 20px; color: #888; }
  </style>

<body>
  <div class="title">🔍 采购查询</div>
  <input type="text" id="searchInput" class="search-box" placeholder="搜索项目/供应商/采购内容" autocomplete="off">
  <div id="results"></div>
  <div id="loading" class="loading" style="display:none;">加载中...</div>

  <script>
    const SUPABASE_URL = 'https://sjgyvhceixyukfgplpts.supabase.co';
    const SUPABASE_KEY = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6InNqZ3l2aGNlaXh5dWtmZ3BscHRzIiwicm9sZSI6ImFub24iLCJpYXQiOjE3NzQzNTg5ODcsImV4cCI6MjA4OTkzNDk4N30.1X0OGO5HGF22SHQlAPwDKsBIRW4DzDHWYm-cl6m-9YY';
    
    let searchTimeout;
    let currentPage = 0;
    let loading = false;
    let hasMore = true;
    let currentKeyword = '';

    const supabase = (path, options = {}) => {
      return fetch(`${SUPABASE_URL}/rest/v1/${path}`, {
        ...options,
        headers: {
          'apikey': SUPABASE_KEY,
          'Authorization': `Bearer ${SUPABASE_KEY}`,
          'Content-Type': 'application/json'
        }
      }).then(res => res.json());
    };

    const loadPurchases = async (reset = false) => {
      if (loading) return;
      loading = true;
      document.getElementById('loading').style.display = 'block';
      
      try {
        if (reset) {
          currentPage = 0;
          hasMore = true;
          document.getElementById('results').innerHTML = '';
        }
        
        const from = currentPage * 20;
        
        let query = `purchases?select=*,projects(name),suppliers(name)&order=purchase_date.desc&limit=20&offset=${from}`;
        
        if (currentKeyword) {
          const projects = await supabase(`projects?select=id&name=ilike.%${encodeURIComponent(currentKeyword)}%`);
          const projectIds = projects.map(p => p.id).join(',');
          
          const suppliers = await supabase(`suppliers?select=id&name=ilike.%${encodeURIComponent(currentKeyword)}%`);
          const supplierIds = suppliers.map(s => s.id).join(',');
          
          const conditions = [];
          if (currentKeyword) conditions.push(`content=ilike.%${encodeURIComponent(currentKeyword)}%`);
          if (projectIds) conditions.push(`project_id=in.(${projectIds})`);
          if (supplierIds) conditions.push(`supplier_id=in.(${supplierIds})`);
          
          if (conditions.length) {
            query = `purchases?select=*,projects(name),suppliers(name)&${conditions.join('&')}&order=purchase_date.desc&limit=20&offset=${from}`;
          }
        }
        
        const data = await supabase(query);
        
        if (!data || !data.length) {
          hasMore = false;
          if (reset) document.getElementById('results').innerHTML = '<div style="text-align:center;padding:20px;color:#888;">暂无数据</div>';
          return;
        }
        
        const purchaseIds = data.map(p => p.id).join(',');
        const payments = await supabase(`transactions?select=purchase_id,amount&type=eq.payment&purchase_id=in.(${purchaseIds})`);
        const paidMap = {};
        payments?.forEach(p => { paidMap[p.purchase_id] = (paidMap[p.purchase_id] || 0) + Math.abs(p.amount); });
        
        const resultsDiv = document.getElementById('results');
        
        data.forEach(p => {
          const paid = paidMap[p.id] || 0;
          const amount = parseFloat(p.amount);
          const unpaid = amount - paid;
          let statusText = '未付款', statusColor = '#e53935';
          if (unpaid <= 0) { statusText = '已付清'; statusColor = '#43a047'; }
          else if (paid > 0) { statusText = '部分付'; statusColor = '#fb8c00'; }
          
          const card = document.createElement('div');
          card.className = 'card';
          card.innerHTML = `
            <div class="flex-between">
              <div><strong>${escapeHtml(p.content || '-')}</strong></div>
              <div class="amount">¥${amount.toLocaleString()}</div>
            </div>
            <div class="flex-between text-muted" style="margin-top: 8px;">
              <div>${p.projects?.name || '-'}</div>
              <div style="color:${statusColor}">${statusText} ${unpaid > 0 ? `(欠¥${unpaid.toLocaleString()})` : ''}</div>
            </div>
            <div class="text-muted">供应商：${p.suppliers?.name || '-'}</div>
            <div class="text-muted">采购日期：${p.purchase_date ? new Date(p.purchase_date).toLocaleDateString() : '-'}</div>
          `;
          resultsDiv.appendChild(card);
        });
        
        currentPage++;
        hasMore = data.length === 20;
        
      } catch (err) {
        console.error(err);
        document.getElementById('results').innerHTML = '<div style="text-align:center;padding:20px;color:#888;">加载失败</div>';
      } finally {
        loading = false;
        document.getElementById('loading').style.display = 'none';
      }
    };
    
    function escapeHtml(str) {
      if (!str) return '';
      return str.replace(/[&<>]/g, function(m) {
        if (m === '&') return '&amp;';
        if (m === '<') return '&lt;';
        if (m === '>') return '&gt;';
        return m;
      });
    }
    
    window.addEventListener('scroll', () => {
      if (loading || !hasMore) return;
      const scrollTop = document.documentElement.scrollTop || document.body.scrollTop;
      const scrollHeight = document.documentElement.scrollHeight;
      const clientHeight = document.documentElement.clientHeight;
      if (scrollTop + clientHeight >= scrollHeight - 100) {
        loadPurchases();
      }
    });
    
    const searchInput = document.getElementById('searchInput');
    searchInput.addEventListener('input', (e) => {
      clearTimeout(searchTimeout);
      searchTimeout = setTimeout(() => {
        currentKeyword = e.target.value;
        loadPurchases(true);
      }, 500);
    });
    
    loadPurchases();
  </script>
